#include <stdio.h>

int main() {
    int a, b, operation, result;

    // --- DATA INPUT ---
    printf("Enter the first integer value (a): ");
    scanf("%d", &a);

    printf("Enter the second integer value (b): ");
    scanf("%d", &b);

    printf("\nChoose a bitwise operation:\n");
    printf("1. AND (&)\n");
    printf("2. OR (|)\n");
    printf("3. XOR (^)\n");
    printf("Your choice: ");
    scanf("%d", &operation);

    printf("\n"); // A space for better readability

    // --- SELECT OPERATION ---
    switch (operation) {
        case 1: // AND Operation
            printf("Operation: AND (&)\n");
            printf("Truth Table (for each bit):\n");
            printf("  - 0 & 0 -> 0\n");
            printf("  - 0 & 1 -> 0\n");
            printf("  - 1 & 0 -> 0\n");
            printf("  - 1 & 1 -> 1\n");
            result = a & b; // Performs the bitwise AND operation
            break;

        case 2: // OR Operation
            printf("Operation: OR (|)\n");
            printf("Truth Table (for each bit):\n");
            printf("  - 0 | 0 -> 0\n");
            printf("  - 0 | 1 -> 1\n");
            printf("  - 1 | 0 -> 1\n");
            printf("  - 1 | 1 -> 1\n");
            result = a | b; // Performs the bitwise OR operation
            break;

        case 3: // XOR Operation
            printf("Operation: XOR (^)\n");
            printf("Truth Table (for each bit):\n");
            printf("  - 0 ^ 0 -> 0\n");
            printf("  - 0 ^ 1 -> 1\n");
            printf("  - 1 ^ 0 -> 1\n");
            printf("  - 1 ^ 1 -> 0\n");
            result = a ^ b; // Performs the bitwise XOR operation
            break;

        default:
            printf("Error: Option '%d' is not valid.\n", operation);
            return 1; // Exits the program with an error code
    }

    // --- RESULT OUTPUT ---
    printf("\n----------------------------------\n");
    printf("The result is: %d\n", result);
    printf("----------------------------------\n");

    return 0; // The program finished successfully
}
